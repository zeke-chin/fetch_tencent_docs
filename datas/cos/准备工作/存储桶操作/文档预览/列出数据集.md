## 列出数据集

#### 功能说明

获取数据集（Dataset）列表。

**注意：**

COS Android SDK 版本需要大于等于 v5.9.30。

#### 示例代码

// APPID

String appid = "1253000000";

DescribeDatasetsRequest request = new DescribeDatasetsRequest(appid);

request.maxresults = 100;// 设置本次返回数据集的最大个数，取值范围为0~200。不设置此参数或者设置为0时，则默认值为100。

request.prefix = "test";// 设置数据集名称前缀。

ciService.describeDatasetsAsync(request, new CosXmlResultListener() {

 @Override

 public void onSuccess(CosXmlRequest request, CosXmlResult cosResult) {

 // result 列出数据集的结果

 // 详细字段请查看api文档或者SDK源码

 DescribeDatasetsResult result = (DescribeDatasetsResult) cosResult;

 }

 @Override

 public void onFail(CosXmlRequest request, CosXmlClientException clientException, CosXmlServiceException serviceException) {

 if (clientException != null) {

 clientException.printStackTrace();

 } else {

 serviceException.printStackTrace();

 }

 }

});
