## 提交网页审核任务

#### 功能说明

用于提交一个网页审核任务。

#### 请求示例

// sdk的引入及初始化cos请参考 https://cloud.tencent.com/document/product/436/31953

function postWebpageAuditing() {

 const config = {

 // 需要替换成您自己的存储桶信息

 Bucket: 'examplebucket-1250000000', // 存储桶，必须

 Region: 'COS_REGION', // 存储桶所在地域，比如ap-beijing，必须

 };

 const host = config.Bucket + '.ci.' + config.Region + '.myqcloud.com';

 const key = 'webpage/auditing'; // 固定值

 const url = `https://${host}/${key}`;

 const body = COS.util.json2xml({

 Request: {

 Input: {

 Url: 'https://cloud.tencent.com/', // 网页文件的链接地址

 },

 Conf: {

 BizType: '',

 }

 }

 });

 cos.request({

 Method: 'POST', // 固定值，必须

 Url: url, // 请求的url，必须

 Key: key, // 固定值，必须

 ContentType: 'application/xml', // 固定值，必须

 Body: body // 请求体参数，必须

 },

 function(err, data){

 if (err) {

 // 处理请求失败

 console.log(err);

 } else {

 // 处理请求成功

 console.log(data.Response);

 }

 });

}

postWebpageAuditing();

 const key = `webpage/auditing/${jobId}`; // 固定值，必须

 const url = `https://${host}/${key}`;

 Method: 'GET', // 固定值，必须
