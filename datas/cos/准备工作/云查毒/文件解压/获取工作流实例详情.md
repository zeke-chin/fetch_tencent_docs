function getWorkflowexecution() {

 // sdk引入以及初始化请参考：https://cloud.tencent.com/document/product/436/8629

 // 需要替换成您自己的存储桶信息

 const key = `workflowexecution/${runId}`; // runId: 实例id;

 const url = `https://${host}/${key}`;

 cos.request(

 {

 Method: 'GET', // 固定值，必须

 Key: key, // 必须

 Url: url, // 请求的url，必须

 },

 function (err, data) {

 if (err) {

 // 处理请求失败

 console.log(err);

 } else {

 // 处理请求成功

 console.log(data.Response);

 }

 }

 );

}
